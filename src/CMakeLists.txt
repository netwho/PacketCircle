# CMakeLists.txt for PacketCircle plugin (Wireshark 4.6.x compatible)
# This plugin creates a custom window showing communication pairs in a circle visualization
# github.com/netwho/PackCircle

include(WiresharkPlugin)

# Plugin name and version info (major minor micro extra)
set_module_info(packetcircle 0 5 1 0)

# Enable Qt MOC and resource compilation
SET(CMAKE_AUTOMOC ON)
SET(CMAKE_AUTORCC ON)

# Find required packages
if(USE_qt6)
    set(qtver "6")
else()
    set(qtver "5")
endif()

find_package(Qt${qtver} REQUIRED COMPONENTS Core Widgets Gui Network)

# Plugin name
set(PLUGIN_NAME packetcircle)

# Source files - dissector/analysis sources
set(DISSECTOR_SRC
    packet_analyzer.c
    circle_plugin.c
    circle_widget.c
    graph_widget.c
    ui_main_window.c
    ui_bridge.cpp
)

# Set C++ for Qt source files (.c files that are actually C++)
set_source_files_properties(
    circle_widget.c
    graph_widget.c
    ui_main_window.c
    ui_bridge.cpp
    PROPERTIES LANGUAGE CXX
)

# glibc 2.38 compatibility shim (Linux only).
# glibc_compat.c is compiled WITHOUT _GNU_SOURCE so sscanf/strtoul inside it
# bind to pre-2.38 glibc symbols. Its weak __isoc23_* symbols satisfy the
# references generated by compiling the rest of the plugin on Ubuntu 24.04,
# eliminating the GLIBC_2.38 dependency from the final .so.
# NOTE: must be appended before PLUGIN_FILES is set below.
if(UNIX AND NOT APPLE)
    list(APPEND DISSECTOR_SRC glibc_compat.c)
    set_source_files_properties(glibc_compat.c PROPERTIES
        COMPILE_OPTIONS  "-U_GNU_SOURCE;-D_POSIX_C_SOURCE=200809L"
        LANGUAGE         C
        SKIP_AUTOGEN     ON
    )
endif()

# Qt resource files
set(PLUGIN_RESOURCES
    packetcircle.qrc
)

# All plugin files (must come after DISSECTOR_SRC is finalised above)
set(PLUGIN_FILES
    plugin.c
    ${DISSECTOR_SRC}
    ${PLUGIN_RESOURCES}
)

set(PLUGIN_HEADERS
    packet_analyzer.h
    circle_widget.h
    graph_widget.h
    circle_plugin.h
    ui_main_window.h
    ui_bridge.h
)

# Include directories
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}
)

# Enable C++ for Qt
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# MSVC: treat source and execution character set as UTF-8.
# Without this MSVC reads .c/.cpp as the system codepage (Windows-1252 on most
# Windows systems), mangling all multi-byte UTF-8 string literals.
if(MSVC)
    add_compile_options(/utf-8)
endif()

# Disable Qt version tagging so the plugin can load with any Qt 6.x runtime
# (Wireshark.app bundles its own Qt which may differ from the build-time Qt)
add_definitions(-DQT_NO_VERSION_TAGGING)

# Set source file properties for plugin.c
set_source_files_properties(
    plugin.c
    PROPERTIES
    SKIP_AUTOGEN ON
    LANGUAGE C
)

set_source_files_properties(
    ${PLUGIN_FILES}
    PROPERTIES
    COMPILE_FLAGS "${WERROR_COMMON_FLAGS}"
)

# Register plugin files (required for Wireshark 4.6.x epan plugin type)
register_plugin_files(plugin.c
    plugin
    ${DISSECTOR_SRC}
)

# Create Wireshark plugin library (epan type for 4.6.x)
add_wireshark_plugin_library(${PLUGIN_NAME} epan)

# Link libraries
target_link_libraries(${PLUGIN_NAME}
    epan
    wiretap
    Qt${qtver}::Core
    Qt${qtver}::Widgets
    Qt${qtver}::Gui
    Qt${qtver}::Network
)

# Install plugin
install_plugin(${PLUGIN_NAME} epan)
